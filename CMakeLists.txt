cmake_minimum_required(VERSION 3.18)
project(portfolio_risk_engine VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(portfolio_core STATIC cpp/src/engine.cpp)
set_target_properties(portfolio_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(portfolio_core PUBLIC cpp/include)
target_link_libraries(portfolio_core PUBLIC Eigen3::Eigen)
option(PORTFOLIO_BUILD_PYTHON "Build Python extension" ON)
option(PORTFOLIO_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(PORTFOLIO_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(portfolio_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(portfolio_core PUBLIC -fsanitize=address,undefined)
endif()
if(PORTFOLIO_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/src/bindings.cpp)
  target_link_libraries(_core PRIVATE portfolio_core)
  install(TARGETS _core LIBRARY DESTINATION portfolio_risk)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_native cpp/tests/test_native.cpp)
  target_link_libraries(test_native PRIVATE portfolio_core)
  add_test(NAME native_engine COMMAND test_native)
endif()
